cmake_minimum_required(VERSION 3.20)
project(FingerprintBench LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
add_compile_definitions(NOMINMAX UNICODE _UNICODE)
set(FRFX_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/third_party/FingerJetFXOSE-master/FingerJetFXOSE/libFRFXLL")
file(GLOB FRFX_SOURCES CONFIGURE_DEPENDS "${FRFX_ROOT}/src/lib/*.cpp")
add_library(FRFXLL SHARED ${FRFX_SOURCES})
target_include_directories(FRFXLL PUBLIC "${FRFX_ROOT}/include" PRIVATE "${FRFX_ROOT}/src/algorithm")
target_compile_options(FRFXLL PRIVATE /W0 /utf-8 /EHsc)
target_compile_definitions(FRFXLL PRIVATE _CRT_SECURE_NO_WARNINGS)
add_library(fingerprint SHARED src/improved.cpp)
add_library(fingerprint_baseline SHARED src/baseline.cpp)
add_library(fingerprint_minutiae SHARED src/minutiae_algorithm.cpp)
target_link_libraries(fingerprint_minutiae PRIVATE FRFXLL)
add_library(fingerprint_graph SHARED src/minutiae_algorithm.cpp)
target_compile_definitions(fingerprint_graph PRIVATE FP_GRAPH_MATCHER)
target_link_libraries(fingerprint_graph PRIVATE FRFXLL)
add_executable(fvc_enroll src/fvc_adapter.cpp)
target_compile_definitions(fvc_enroll PRIVATE FVC_ENROLL)
set_target_properties(fvc_enroll PROPERTIES OUTPUT_NAME enroll)
target_link_libraries(fvc_enroll PRIVATE fingerprint_minutiae windowscodecs ole32)
add_executable(fvc_match src/fvc_adapter.cpp)
set_target_properties(fvc_match PROPERTIES OUTPUT_NAME match)
target_link_libraries(fvc_match PRIVATE fingerprint_minutiae windowscodecs ole32)
add_executable(fp_worker src/worker.cpp)
target_link_libraries(fp_worker PRIVATE windowscodecs ole32)
add_executable(fp_eval src/evaluator.cpp)
add_executable(fp_task src/task.cpp)
target_link_libraries(fp_task PRIVATE windowscodecs ole32)
add_executable(fp_bench src/bench.cpp)
target_link_libraries(fp_bench PRIVATE bcrypt windowscodecs ole32)
add_executable(fp_protocol src/protocol.cpp)
target_link_libraries(fp_protocol PRIVATE bcrypt windowscodecs ole32)
add_executable(fp_calibrate src/calibrate.cpp)
target_link_libraries(fp_calibrate PRIVATE bcrypt)
add_executable(fp_plan src/planner.cpp)
add_executable(fp_inspect src/inspect.cpp)
target_link_libraries(fp_inspect PRIVATE bcrypt windowscodecs ole32)
foreach(asset index.html viewer.js)
  configure_file("assets/inspect/${asset}" "inspect-assets/${asset}" COPYONLY)
  foreach(config Debug Release RelWithDebInfo MinSizeRel)
    configure_file("assets/inspect/${asset}" "${config}/inspect-assets/${asset}" COPYONLY)
  endforeach()
endforeach()
add_executable(fp_gui WIN32 src/gui.cpp)
target_link_libraries(fp_gui PRIVATE comctl32 ole32 shell32)
enable_testing()
add_executable(fp_metrics_test tests/metrics_test.cpp)
add_test(NAME metrics COMMAND fp_metrics_test)
add_executable(fp_minutiae_test tests/minutiae_test.cpp)
add_test(NAME minutiae COMMAND fp_minutiae_test)
add_executable(fp_graph_test tests/graph_test.cpp)
add_test(NAME graph_matcher COMMAND fp_graph_test)
add_executable(fp_abi_test tests/abi_test.cpp)
add_dependencies(fp_abi_test fingerprint_minutiae)
add_test(NAME algorithm_abi COMMAND fp_abi_test $<TARGET_FILE:fingerprint_minutiae>)
foreach(mode fail hang crash badsize badscore comparehang comparecrash overrun inputwrite)
  add_library(fixture_${mode} SHARED src/fixture.cpp)
  target_compile_definitions(fixture_${mode} PRIVATE FIXTURE_${mode})
endforeach()
foreach(target fingerprint fingerprint_baseline fingerprint_minutiae fingerprint_graph fp_graph_test fvc_enroll fvc_match fp_worker fp_eval fp_task fp_bench fp_plan fp_protocol fp_calibrate fp_inspect fp_gui fp_metrics_test fp_minutiae_test fp_abi_test fixture_fail fixture_hang fixture_crash fixture_badsize fixture_badscore fixture_comparehang fixture_comparecrash fixture_overrun fixture_inputwrite)
  target_compile_options(${target} PRIVATE /W4 /utf-8 /EHsc)
endforeach()
