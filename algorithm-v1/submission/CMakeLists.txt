cmake_minimum_required(VERSION 3.20)
project(FingerprintV1Submission LANGUAGES CXX)
if(NOT MSVC)
    message(FATAL_ERROR "Windows/MSVC required")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(NOMINMAX UNICODE _UNICODE)
set(FRFX_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../third_party/FingerJetFXOSE-master/FingerJetFXOSE/libFRFXLL")
file(GLOB FRFX_SOURCES CONFIGURE_DEPENDS "${FRFX_ROOT}/src/lib/*.cpp")
add_library(FRFXLL SHARED ${FRFX_SOURCES})
target_include_directories(FRFXLL PUBLIC "${FRFX_ROOT}/include" PRIVATE "${FRFX_ROOT}/src/algorithm")
target_compile_definitions(FRFXLL PRIVATE _CRT_SECURE_NO_WARNINGS _SILENCE_NONFLOATING_COMPLEX_DEPRECATION_WARNING)
target_compile_options(FRFXLL PRIVATE /W0 /utf-8 /EHsc)
foreach(target IN ITEMS fingerprint_algorithm fingerprint_algorithm_eval)
    add_library(${target} SHARED src/algorithm.cpp src/algorithm.def)
    target_include_directories(${target} PRIVATE "${FRFX_ROOT}/include")
    target_compile_options(${target} PRIVATE /W4 /utf-8 /EHsc)
    # Deliberately NO import library link: sibling FRFXLL is loaded on demand.
endforeach()
target_compile_definitions(fingerprint_algorithm_eval PRIVATE FP_THREE_DECIMAL_SCORE)
enable_testing()
add_executable(fp_core_tests tests/core_tests.cpp)
target_include_directories(fp_core_tests PRIVATE src)
target_compile_options(fp_core_tests PRIVATE /W4 /utf-8 /EHsc)
add_test(NAME core_contract COMMAND fp_core_tests)
add_executable(fp_probe tests/probe.cpp)
target_include_directories(fp_probe PRIVATE src)
target_compile_options(fp_probe PRIVATE /W4 /utf-8 /EHsc)
add_library(throwing_extractor SHARED tests/throwing_extractor.cpp tests/throwing_extractor.def)
target_compile_options(throwing_extractor PRIVATE /W4 /utf-8 /EHsc)
add_executable(fp_compare examples/compare.cpp)
target_compile_options(fp_compare PRIVATE /W4 /utf-8 /EHsc)
