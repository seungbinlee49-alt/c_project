cmake_minimum_required(VERSION 3.20)
project(FingerprintAlgorithmV1 LANGUAGES CXX)
if(NOT MSVC)
  message(FATAL_ERROR "Windows / MSVC is required")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_definitions(NOMINMAX UNICODE _UNICODE)
set(FRFX_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../third_party/FingerJetFXOSE-master/FingerJetFXOSE/libFRFXLL")
file(GLOB FRFX_SOURCES CONFIGURE_DEPENDS "${FRFX_ROOT}/src/lib/*.cpp")
add_library(FRFXLL SHARED ${FRFX_SOURCES})
target_include_directories(FRFXLL PUBLIC "${FRFX_ROOT}/include" PRIVATE "${FRFX_ROOT}/src/algorithm")
target_compile_options(FRFXLL PRIVATE /W0 /utf-8 /EHsc)
target_compile_definitions(FRFXLL PRIVATE _CRT_SECURE_NO_WARNINGS)
add_library(fingerprint_algorithm SHARED src/minutiae_algorithm.cpp)
target_compile_definitions(fingerprint_algorithm PRIVATE FP_ORIENTATION_TEMPLATE FP_CONTRAST_NORMALIZATION)
target_compile_options(fingerprint_algorithm PRIVATE /W4 /utf-8 /EHsc)
target_link_libraries(fingerprint_algorithm PRIVATE FRFXLL)
add_executable(fp_compare examples/compare.cpp)
target_compile_options(fp_compare PRIVATE /W4 /utf-8 /EHsc)
