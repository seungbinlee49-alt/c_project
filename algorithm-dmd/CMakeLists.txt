cmake_minimum_required(VERSION 3.20)
project(DmdCpu64Release LANGUAGES CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
 message(FATAL_ERROR "This release requires MSVC x64")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded")
set(FJ_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../third_party/FingerJetFXOSE-master/FingerJetFXOSE/libFRFXLL")
add_library(fingerprint_dmd_cpu64 SHARED src/algorithm.cpp)
target_include_directories(fingerprint_dmd_cpu64 PRIVATE third_party/onnxruntime/include "${FJ_ROOT}/include")
target_compile_definitions(fingerprint_dmd_cpu64 PRIVATE NOMINMAX UNICODE _UNICODE ORT_API_MANUAL_INIT)
target_compile_options(fingerprint_dmd_cpu64 PRIVATE /O2 /utf-8 /EHsc /W4)
target_link_libraries(fingerprint_dmd_cpu64 PRIVATE bcrypt)
add_executable(dmd_probe src/probe.cpp)
target_compile_definitions(dmd_probe PRIVATE NOMINMAX UNICODE _UNICODE)
target_compile_options(dmd_probe PRIVATE /O2 /utf-8 /EHsc /W4)

# Optional rebuild of the replaceable LGPL dependency from the bundled source.
option(BUILD_FINGERJET_FROM_SOURCE "Build the bundled FingerJet DLL" OFF)
if(BUILD_FINGERJET_FROM_SOURCE)
 set(FRFX_ROOT "${FJ_ROOT}")
 file(GLOB FRFX_SOURCES CONFIGURE_DEPENDS "${FRFX_ROOT}/src/lib/*.cpp")
 add_library(FRFXLL SHARED ${FRFX_SOURCES})
 target_include_directories(FRFXLL PRIVATE "${FRFX_ROOT}/include" "${FRFX_ROOT}/src/algorithm")
 target_compile_definitions(FRFXLL PRIVATE _CRT_SECURE_NO_WARNINGS _SILENCE_NONFLOATING_COMPLEX_DEPRECATION_WARNING)
 target_compile_options(FRFXLL PRIVATE /W0 /utf-8 /EHsc)
endif()
